-- Lưu trữ ESP
local espData = {}
local espLock = false

local function clearESP()
    espLock = true
    for _, data in pairs(espData) do
        if data.box and data.box.Parent then data.box:Destroy() end
        if data.billboard and data.billboard.Parent then data.billboard:Destroy() end
    end
    espData = {}
    espLock = false
end

-- Tạo ESP (hộp màu + tên nếu bật)
local function addESP(obj, color, name)
    if not obj or not obj.Parent or espLock then return end
    local box = Instance.new("BoxHandleAdornment")
    box.Size = obj.Size + Vector3.new(0.5, 0.5, 0.5)
    box.Color3 = color
    box.Transparency = 0.3
    box.AlwaysOnTop = true
    box.ZIndex = 999
    box.Adornee = obj
    box.Parent = obj

    local billboard = nil
    if settings.espNames and name then
        local target = obj
        if obj:IsA("Model") then
            for _, p in pairs(obj:GetChildren()) do
                if p:IsA("BasePart") then target = p; break end
            end
        end
        if target and target:IsA("BasePart") then
            billboard = Instance.new("BillboardGui")
            billboard.Size = UDim2.new(0, 150, 0, 25)
            billboard.AlwaysOnTop = true
            billboard.Adornee = target
            billboard.Parent = target

            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.Text = name
            label.TextColor3 = Color3.new(1, 1, 1)
            label.TextSize = 13
            label.Font = Enum.Font.SourceSansBold
            label.TextStrokeTransparency = 0.3
            label.TextStrokeColor3 = Color3.new(0, 0, 0)
            label.Parent = billboard
        end
    end
    table.insert(espData, {box = box, billboard = billboard})
end

local function makeToggle(text, y, key)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -8, 0, 24)
    frame.Position = UDim2.new(0, 4, 0, y)
    frame.BackgroundTransparency = 1
    frame.Parent = main

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.SourceSans
    label.Parent = frame

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.32, 0, 1, -2)
    btn.Position = UDim2.new(0.68, 0, 0, 1)
    btn.BackgroundColor3 = Color3.new(0.1, 0, 0)
    btn.BorderColor3 = Color3.new(1, 0, 0)
    btn.BorderSizePixel = 1
    btn.Text = "OFF"
    btn.TextColor3 = Color3.new(1, 0, 0)
    btn.TextSize = 11
    btn.Font = Enum.Font.SourceSansBold
    btn.Parent = frame

    btn.MouseButton1Click:Connect(function()
        settings[key] = not settings[key]
        btn.Text = settings[key] and "ON" or "OFF"
        btn.TextColor3 = settings[key] and Color3.new(0, 1, 0) or Color3.new(1, 0, 0)
        if not settings[key] and string.find(key, "esp") then clearESP() end
    end)
end

-- === Main Loop ===
local hrp, hum
local frameCount = 0

RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end

    hum.WalkSpeed = settings.speed

    -- ESP refresh mỗi 2 frame để giảm tải
    frameCount = frameCount + 1
    if frameCount % 2 == 0 then
        if settings.espPlayers or settings.espMonsters or settings.espItems or settings.espChests or settings.espCoins then
            clearESP()

            -- Players (Đỏ)
            if settings.espPlayers then
                for _, plr in pairs(Players:GetPlayers()) do
                    if plr ~= LocalPlayer and plr.Character then
                        for _, part in pairs(plr.Character:GetDescendants()) do
                            if part:IsA("BasePart") then addESP(part, Color3.new(1, 0, 0)) end
                        end
                    end
                end
            end

            -- Monsters (Cam) + tên
            if settings.espMonsters then
                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj:IsA("Model") and (obj.Name:lower():find("monster") or obj.Name:lower():find("enemy") or obj.Name:lower():find("boss") or obj.Name:lower():find("npc")) then
                        local name = obj.Name
                        for _, part in pairs(obj:GetDescendants()) do
                            if part:IsA("BasePart") then addESP(part, Color3.new(1, 0.5, 0), name) end
                        end
                    end
                end
            end

            -- Items (Xanh dương) + tên chi tiết
            if settings.espItems then
                for _, item in pairs(workspace:GetDescendants()) do
                    if item:IsA("Tool") and item:FindFirstChild("Handle") then
                        local name = item.Name
                        addESP(item.Handle, Color3.new(0, 0.8, 1), name)
                    end
                end
            end

            -- Chests/Doors (Xanh lá) + tên
            if settings.espChests then
                for _, part in pairs(workspace:GetDescendants()) do
                    if part:IsA("BasePart") then
                        local n = part.Name:lower()
                        if n:find("chest") or n:find("door") or n:find("crate") or n:find("vault") or n:find("cabinet") then
                            addESP(part, Color3.new(0, 1, 0), part.Name)
                        end
                    end
                end
            end

            -- Coins (Vàng) + tên
            if settings.espCoins then
                for _, part in pairs(workspace:GetDescendants()) do
                    if part:IsA("BasePart") then
                        local n = part.Name:lower()
                        if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") or n:find("diamond") then
                            addESP(part, Color3.new(1, 1, 0), part.Name)
                        end
                    end
                end
            end
        else
            clearESP()
        end
    end

    -- === AUTO FUNCTIONS (Aura) ===
    local pos = hrp.Position
    local range = settings.range

    if settings.autoCollect then
        for _, tool in pairs(workspace:GetDescendants()) do
            if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                local dist = (tool.Handle.Position - pos).Magnitude
                if dist < range then
                    firetouchinterest(hrp, tool.Handle, 0)
                    firetouchinterest(hrp, tool.Handle, 1)
                end
            end
        end
    end

    if settings.autoOpen then
        for _, detector in pairs(workspace:GetDescendants()) do
            if detector:IsA("ClickDetector") and detector.Parent then
                local p = detector.Parent
                local target = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if target then
                    local dist = (target.Position - pos).Magnitude
                    if dist < range then
                        detector.MouseClick:Fire(LocalPlayer)
                    end
                end
            end
        end
        -- Mở tủ đồ, két sắt (mô phỏng bằng cách kích hoạt ProximityPrompt nếu có)
        for _, prompt in pairs(workspace:GetDescendants()) do
            if prompt:IsA("ProximityPrompt") and prompt.Parent then
                local dist = (prompt.Parent.Position - pos).Magnitude
                if dist < range then
                    prompt.HoldEnd:Fire(LocalPlayer)
                end
            end
        end
    end

    if settings.autoLoot then
        for _, part in pairs(workspace:GetDescendants()) do
            if part:IsA("BasePart") then
                local n = part.Name:lower()
                if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") then
                    local dist = (part.Position - pos).Magnitude
                    if dist < range then
                        firetouchinterest(hrp, part, 0)
                        firetouchinterest(hrp, part, 1)
                    end
                end
            end
        end
    end
end)

-- Toggle menu bằng F1
UserInputService.InputBegan:Connect(function(inp, gp)
    if gp then return end
    if inp.KeyCode == Enum.KeyCode.F1 then
        main.Visible = not main.Visible
    end
end)
