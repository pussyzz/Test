local espObjects = {}
local function clearESP()
    for _, obj in pairs(espObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    espObjects = {}
end

local function createESP(part, color, name)
    if not part or not part.Parent then return end
    local highlight = Instance.new("Highlight")
    highlight.FillColor = color
    highlight.FillTransparency = 0.4
    highlight.OutlineColor = color
    highlight.OutlineTransparency = 0.2
    highlight.Adornee = part
    highlight.Parent = part
    table.insert(espObjects, highlight)

    if settings.espNames and name then
        local bill = Instance.new("BillboardGui")
        bill.Size = UDim2.new(0, 120, 0, 20)
        bill.AlwaysOnTop = true
        bill.Adornee = part
        bill.Parent = part
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.Text = name
        label.TextColor3 = Color3.new(1,1,1)
        label.TextSize = 12
        label.Font = Enum.Font.SourceSansBold
        label.TextStrokeTransparency = 0.3
        label.Parent = bill
        table.insert(espObjects, bill)
    end
end

local function makeToggle(text, y, key)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1,-8,0,24)
    frame.Position = UDim2.new(0,4,0,y)
    frame.BackgroundTransparency = 1
    frame.Parent = main
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6,0,1,0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.new(1,1,1)
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.SourceSans
    label.Parent = frame
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.32,0,1,-2)
    btn.Position = UDim2.new(0.68,0,0,1)
    btn.BackgroundColor3 = Color3.new(0.1,0,0)
    btn.BorderColor3 = Color3.new(1,0,0)
    btn.BorderSizePixel = 1
    btn.Text = "OFF"
    btn.TextColor3 = Color3.new(1,0,0)
    btn.TextSize = 11
    btn.Font = Enum.Font.SourceSansBold
    btn.Parent = frame
    btn.MouseButton1Click:Connect(function()
        settings[key] = not settings[key]
        btn.Text = settings[key] and "ON" or "OFF"
        btn.TextColor3 = settings[key] and Color3.new(0,1,0) or Color3.new(1,0,0)
        if not settings[key] and string.find(key,"esp") then clearESP() end
    end)
end

-- Main Loop
local hrp, hum
local scanCount = 0
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    hum.WalkSpeed = settings.speed

    scanCount = scanCount + 1
    if scanCount % 3 == 0 and (settings.espPlayers or settings.espMonsters or settings.espItems or settings.espChests or settings.espCoins) then
        clearESP()
        if settings.espPlayers then
            for _, plr in pairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Character then
                    for _, part in pairs(plr.Character:GetDescendants()) do
                        if part:IsA("BasePart") then createESP(part, Color3.new(1,0,0)) end
                    end
                end
            end
        end
        if settings.espMonsters then
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and (obj.Name:lower():find("monster") or obj.Name:lower():find("enemy") or obj.Name:lower():find("boss") or obj.Name:lower():find("npc")) then
                    for _, part in pairs(obj:GetDescendants()) do
                        if part:IsA("BasePart") then createESP(part, Color3.new(1,0.5,0), obj.Name) end
                    end
                end
            end
        end
        if settings.espItems then
            for _, item in pairs(workspace:GetDescendants()) do
                if item:IsA("Tool") and item:FindFirstChild("Handle") then
                    createESP(item.Handle, Color3.new(0,0.8,1), item.Name)
                end
            end
        end
        if settings.espChests then
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") then
                    local n = part.Name:lower()
                    if n:find("chest") or n:find("door") or n:find("crate") or n:find("vault") or n:find("cabinet") or n:find("locker") then
                        createESP(part, Color3.new(0,1,0), part.Name)
                    end
                end
            end
        end
        if settings.espCoins then
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") then
                    local n = part.Name:lower()
                    if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") or n:find("diamond") then
                        createESP(part, Color3.new(1,1,0), part.Name)
                    end
                end
            end
        end
    end

    -- Auto Aura (spawn ít lần)
    if settings.autoCollect or settings.autoOpen or settings.autoLoot then
        local pos = hrp.Position
        local range = settings.range
        for _, obj in pairs(workspace:GetDescendants()) do
            if settings.autoCollect and obj:IsA("Tool") and obj:FindFirstChild("Handle") then
                if (obj.Handle.Position - pos).Magnitude < range then
                    firetouchinterest(hrp, obj.Handle, 0)
                    firetouchinterest(hrp, obj.Handle, 1)
                end
            end
            if settings.autoOpen and obj:IsA("ClickDetector") and obj.Parent then
                local p = obj.Parent
                local target = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if target and (target.Position - pos).Magnitude < range then
                    obj.MouseClick:Fire(LocalPlayer)
                end
            end
            if settings.autoOpen and obj:IsA("ProximityPrompt") and obj.Parent then
                if (obj.Parent.Position - pos).Magnitude < range then
                    obj.HoldEnd:Fire(LocalPlayer)
                end
            end
            if settings.autoLoot and obj:IsA("BasePart") then
                local n = obj.Name:lower()
                if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") then
                    if (obj.Position - pos).Magnitude < range then
                        firetouchinterest(hrp, obj, 0)
                        firetouchinterest(hrp, obj, 1)
                    end
                end
            end
        end
    end
end)

UserInputService.InputBegan:Connect(function(inp,gp)

    if inp.KeyCode == Enum.KeyCode.F1 then
        main.Visible = not main.Visible
    end
end)
