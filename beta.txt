-- ESP Storage
local espList = {}
local espLock = false

local function clearESP()
    espLock = true
    for i = #espList, 1, -1 do
        local v = espList[i]
        if v and v.Parent then v:Destroy() end
    end
    espList = {}
    espLock = false
end

local function addESP(obj, color)
    if not obj or not obj.Parent or espLock then return end
    local box = Instance.new("BoxHandleAdornment")
    box.Size = obj.Size + Vector3.new(0.5, 0.5, 0.5)
    box.Color3 = color
    box.Transparency = 0.35
    box.AlwaysOnTop = true
    box.ZIndex = 999
    box.Adornee = obj
    box.Parent = obj
    table.insert(espList, box)
end

local function makeToggle(text, y, key)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -8, 0, 22)
    frame.Position = UDim2.new(0, 4, 0, y)
    frame.BackgroundTransparency = 1
    frame.Parent = main
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.65, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.SourceSans
    label.Parent = frame
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.3, 0, 1, -2)
    btn.Position = UDim2.new(0.7, 0, 0, 1)
    btn.BackgroundColor3 = Color3.new(0.1, 0, 0)
    btn.BorderColor3 = Color3.new(1, 0, 0)
    btn.BorderSizePixel = 1
    btn.Text = "OFF"
    btn.TextColor3 = Color3.new(1, 0, 0)
    btn.TextSize = 11
    btn.Font = Enum.Font.SourceSansBold
    btn.Parent = frame
    
    btn.MouseButton1Click:Connect(function()
        settings[key] = not settings[key]
        btn.Text = settings[key] and "ON" or "OFF"
        btn.TextColor3 = settings[key] and Color3.new(0, 1, 0) or Color3.new(1, 0, 0)
        if not settings[key] and string.find(key, "esp") then clearESP() end
    end)
end

-- Main loop (optimized)
local hrp, hum
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    
    hum.WalkSpeed = settings.speed
    
    -- ESP (only if any enabled)
    if settings.espPlayers or settings.espMonsters or settings.espItems or settings.espChests or settings.espCoins then
        clearESP()
        
        if settings.espPlayers then
            for _, plr in pairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and plr.Character then
                    for _, part in pairs(plr.Character:GetDescendants()) do
                        if part:IsA("BasePart") then addESP(part, Color3.new(1, 0, 0)) end
                    end
                end
            end
        end
        
        if settings.espMonsters then
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and (obj.Name:lower():find("monster") or obj.Name:lower():find("enemy") or obj.Name:lower():find("boss") or obj.Name:lower():find("npc")) then
                    for _, part in pairs(obj:GetDescendants()) do
                        if part:IsA("BasePart") then addESP(part, Color3.new(1, 0.5, 0)) end
                    end
                end
            end
        end
        
        if settings.espItems then
            for _, item in pairs(workspace:GetDescendants()) do
                if item:IsA("Tool") and item:FindFirstChild("Handle") then
                    addESP(item.Handle, Color3.new(0, 1, 1))
                end
            end
        end
        
        if settings.espChests then
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") and (part.Name:lower():find("chest") or part.Name:lower():find("door") or part.Name:lower():find("crate") or part.Name:lower():find("vault")) then
                    addESP(part, Color3.new(0, 1, 0))
                end
            end
        end
        
        if settings.espCoins then
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") and (part.Name:lower():find("coin") or part.Name:lower():find("money") or part.Name:lower():find("gold") or part.Name:lower():find("gem") or part.Name:lower():find("diamond")) then
                    addESP(part, Color3.new(1, 1, 0))
                end
            end
        end
    else
        clearESP()
    end
    
    -- Auto functions (fire only when needed)
    if settings.autoCollect then
        for _, tool in pairs(workspace:GetDescendants()) do
            if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                local dist = (tool.Handle.Position - hrp.Position).Magnitude
                if dist < 15 then
                    firetouchinterest(hrp, tool.Handle, 0)
                    firetouchinterest(hrp, tool.Handle, 1)
                end
            end
        end
    end
    
    if settings.autoOpen then
        for _, detector in pairs(workspace:GetDescendants()) do
            if detector:IsA("ClickDetector") and detector.Parent then
                local p = detector.Parent
                local pos = p:IsA("BasePart") and p.Position or (p:FindFirstChild("Handle") and p.Handle.Position)
                if pos then
                    local dist = (pos - hrp.Position).Magnitude
                    if dist < 12 then
                        detector.MouseClick:Fire(LocalPlayer)
                    end
                end
            end
        end
    end
    
    if settings.autoLoot then
        for _, part in pairs(workspace:GetDescendants()) do
            if part:IsA("BasePart") and (part.Name:lower():find("coin") or part.Name:lower():find("money") or part.Name:lower():find("gold") or part.Name:lower():find("gem")) then
                local dist = (part.Position - hrp.Position).Magnitude
                if dist < 12 then
                    firetouchinterest(hrp, part, 0)
                    firetouchinterest(hrp, part, 1)
                end
            end
        end
    end
end)
