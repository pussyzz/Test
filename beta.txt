-- ESP storage
local espObjects = {}
local function clearESP()
    for _, obj in pairs(espObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    espObjects = {}
end

local function createESP(part, color, name)
    if not part or not part.Parent then return end
    local highlight = Instance.new("Highlight")
    highlight.FillColor = color
    highlight.FillTransparency = 0.35
    highlight.OutlineColor = color
    highlight.OutlineTransparency = 0.2
    highlight.Adornee = part
    highlight.Parent = part
    table.insert(espObjects, highlight)

    if name then
        local bill = Instance.new("BillboardGui")
        bill.Size = UDim2.new(0, 140, 0, 22)
        bill.AlwaysOnTop = true
        bill.Adornee = part
        bill.Parent = part
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.Text = name
        label.TextColor3 = Color3.new(1,1,1)
        label.TextSize = 12
        label.Font = Enum.Font.SourceSansBold
        label.TextStrokeTransparency = 0.2
        label.Parent = bill
        table.insert(espObjects, bill)
    end
end

-- Main Loop
local hrp, hum
local scan = 0
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    hum.WalkSpeed = settings.speed

    -- ESP (chỉ items đặc biệt + quái)
    if settings.espEnabled then
        scan = scan + 1
        if scan % 3 == 0 then
            clearESP()
            -- Quái (cam)
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and (obj.Name:lower():find("monster") or obj.Name:lower():find("enemy") or obj.Name:lower():find("boss") or obj.Name:lower():find("npc")) then
                    for _, part in pairs(obj:GetDescendants()) do
                        if part:IsA("BasePart") then createESP(part, Color3.new(1,0.5,0), obj.Name) end
                    end
                end
            end
            -- Items: key, rương, note, két sắt, tủ, locker, cabinet, vault (xanh dương)
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") then
                    local n = part.Name:lower()
                    if n:find("key") or n:find("ruong") or n:find("chest") or n:find("note") or n:find("giay") or n:find("két") or n:find("sắt") or n:find("locker") or n:find("cabinet") or n:find("vault") or n:find("door") then
                        createESP(part, Color3.new(0,0.8,1), part.Name)
                    end
                end
            end
            -- Tools (item cầm tay)
            for _, tool in pairs(workspace:GetDescendants()) do
                if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                    createESP(tool.Handle, Color3.new(0,0.8,1), tool.Name)
                end
            end
        end
    else
        clearESP()
    end

    -- Aura: nhặt xu + mở cửa/tủ
    if settings.auraEnabled and hrp then
        local pos = hrp.Position
        for _, obj in pairs(workspace:GetDescendants()) do
            -- Nhặt xu (coin, money, gold, gem)
            if obj:IsA("BasePart") then
                local n = obj.Name:lower()
                if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") or n:find("diamond") then
                    if (obj.Position - pos).Magnitude < 18 then
                        firetouchinterest(hrp, obj, 0)
                        firetouchinterest(hrp, obj, 1)
                    end
                end
            end
            -- Mở cửa, tủ, két (ClickDetector)
            if obj:IsA("ClickDetector") and obj.Parent then
                local p = obj.Parent
                local target = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if target and (target.Position - pos).Magnitude < 15 then
                    obj.MouseClick:Fire(LocalPlayer)
                end
            end
            -- Mở bằng ProximityPrompt
            if obj:IsA("ProximityPrompt") and obj.Parent then
                if (obj.Parent.Position - pos).Magnitude < 15 then
                    obj.HoldEnd:Fire(LocalPlayer)
                end
            end
        end
    end
end)

-- F1 mở/tắt menu (thu nhỏ/mở rộng)
UserInputService.InputBegan:Connect(function(inp, gp)
    if gp then return end
    if inp.KeyCode == Enum.KeyCode.F1 then
        if main.Visible then
            toggleExpand()
        else
            main.Visible = true
        end
    end
end)
