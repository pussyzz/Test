-- ESP storage
local espObjects = {}
local function clearESP()
    for _, obj in pairs(espObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    espObjects = {}
end

local function createESP(part, color, name)
    if not part or not part.Parent then return end
    local h = Instance.new("Highlight")
    h.FillColor = color
    h.FillTransparency = 0.3
    h.OutlineColor = color
    h.OutlineTransparency = 0.1
    h.Adornee = part
    h.Parent = part
    table.insert(espObjects, h)
    if name then
        local bill = Instance.new("BillboardGui")
        bill.Size = UDim2.new(0, 150, 0, 24)
        bill.AlwaysOnTop = true
        bill.Adornee = part
        bill.Parent = part
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(1,0,1,0)
        lbl.BackgroundTransparency = 1
        lbl.Text = name
        lbl.TextColor3 = Color3.new(1,1,1)
        lbl.TextSize = 13
        lbl.Font = Enum.Font.SourceSansBold
        lbl.TextStrokeTransparency = 0.2
        lbl.Parent = bill
        table.insert(espObjects, bill)
    end
end

-- Main Loop
local hrp, hum
local scan = 0
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    hum.WalkSpeed = settings.speed

    local pos = hrp.Position
    local espRange = 300  -- Giới hạn tuyệt đối, không quét xa hơn
    local auraRange = 25

    -- ESP - Chỉ quét 1 lần mỗi 5 frame để giảm CPU
    if settings.esp then
        scan = scan + 1
        if scan % 5 == 0 then
            clearESP()
            
            -- 1. QUÁI PETAPETA (Fix lỗi mất định vị)
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") then
                    local name = obj.Name:lower()
                    -- Thêm "quái", "peta" để bắt chính xác
                    if name:find("petapeta") or name:find("peta") or name:find("monster") or name:find("enemy") or name:find("boss") or name:find("npc") or name:find("quái") then
                        local primary = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Head") or obj:FindFirstChild("Handle")
                        if primary and (primary.Position - pos).Magnitude < espRange then
                            for _, part in pairs(obj:GetDescendants()) do
                                if part:IsA("BasePart") then 
                                    createESP(part, Color3.new(1,0.5,0), "Petapeta") 
                                end
                            end
                        end
                    end
                elseif obj:IsA("BasePart") then
                    local name = obj.Name:lower()
                    if name:find("petapeta") or name:find("peta") or name:find("quái") then
                        if (obj.Position - pos).Magnitude < espRange then
                            createESP(obj, Color3.new(1,0.5,0), "Petapeta")
                        end
                    end
                end
            end

            -- 2. ITEMS: Tờ giấy có mã, Bùa Ofuda, Két sắt, Hộp, Tủ khóa
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") then
                    -- Kiểm tra khoảng cách NGAY LẬP TỨC để bỏ qua object xa -> chống lag
                    local dist = (part.Position - pos).Magnitude
                    if dist > espRange then
                        continue
                    end
                    
                    local n = part.Name:lower()
                    local isTarget = n:find("giấy") or n:find("giay") or n:find("mã") or n:find("ma") or 
                                     n:find("bùa") or n:find("bua") or n:find("ofuda") or 
                                     n:find("két") or n:find("ket") or n:find("safe") or 
                                     n:find("hộp") or n:find("hop") or n:find("chest") or 
                                     n:find("tủ") or n:find("tu") or n:find("locker") or n:find("cabinet") or
                                     n:find("key") or n:find("khoá") or n:find("khoa") or n:find("khóa")
                    
                    -- Loại trừ cửa để không bị loạn
                    if isTarget and not n:find("door") and not n:find("cửa") and not n:find("cua") then
                        local displayName = "Item"
                        if n:find("giấy") or n:find("giay") or n:find("mã") or n:find("ma") then 
                            displayName = "Tờ giấy có mã"
                        elseif n:find("bùa") or n:find("bua") or n:find("ofuda") then 
                            displayName = "Bùa Ofuda"
                        elseif n:find("két") or n:find("ket") or n:find("safe") or n:find("chest") then 
                            displayName = "Két sắt"
                        elseif n:find("hộp") or n:find("hop") then 
                            displayName = "Hộp"
                        elseif n:find("key") or n:find("khoá") or n:find("khoa") or n:find("khóa") then 
                            displayName = "Chìa khóa"
                        elseif n:find("tủ") or n:find("tu") or n:find("locker") or n:find("cabinet") then 
                            displayName = "Tủ"
                        end
                        createESP(part, Color3.new(0,0.8,1), displayName)
                    end
                end
            end

            -- 3. TOOLS (đồ cầm tay)
            for _, tool in pairs(workspace:GetDescendants()) do
                if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                    local dist = (tool.Handle.Position - pos).Magnitude
                    if dist < espRange then
                        createESP(tool.Handle, Color3.new(0,0.8,1), tool.Name)
                    end
                end
            end
        end
    else
        clearESP()
    end

    -- AURA (giữ nguyên, tối ưu nhẹ)
    if settings.aura and hrp then
        for _, obj in pairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                local n = obj.Name:lower()
                if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") then
                    if (obj.Position - pos).Magnitude < auraRange then
                        firetouchinterest(hrp, obj, 0)
                        firetouchinterest(hrp, obj, 1)
                    end
                end
            end
            if obj:IsA("ClickDetector") and obj.Parent then
                local p = obj.Parent
                local target = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if target then
                    local n = target.Name:lower()
                    if not n:find("door") and not n:find("cửa") and not n:find("cua") then
                        if (target.Position - pos).Magnitude < auraRange then
                            obj.MouseClick:Fire(LocalPlayer)
                        end
                    end
                end
            end
            if obj:IsA("ProximityPrompt") and obj.Parent then
                local p = obj.Parent
                local posObj = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if posObj then
                    local n = posObj.Name:lower()
                    if not n:find("door") and not n:find("cửa") and not n:find("cua") then
                        if (posObj.Position - pos).Magnitude < auraRange then
                            obj.HoldEnd:Fire(LocalPlayer)
                        end
                    end
                end
            end
        end
    end
end)

-- F1 toggle
UserInputService.InputBegan:Connect(function(inp, gp)
    if gp then return end
    if inp.KeyCode == Enum.KeyCode.F1 then
        if main.Visible then
            toggleExpand()
        else
            main.Visible = true
        end
    end
end)
