-- ESP storage
local espObjects = {}
local function clearESP()
    for _, obj in pairs(espObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    espObjects = {}
end

local function createESP(part, color, name)
    if not part or not part.Parent then return end
    local h = Instance.new("Highlight")
    h.FillColor = color
    h.FillTransparency = 0.35
    h.OutlineColor = color
    h.OutlineTransparency = 0.1
    h.Adornee = part
    h.Parent = part
    table.insert(espObjects, h)
    if name then
        local bill = Instance.new("BillboardGui")
        bill.Size = UDim2.new(0, 130, 0, 22)
        bill.AlwaysOnTop = true
        bill.Adornee = part
        bill.Parent = part
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(1,0,1,0)
        lbl.BackgroundTransparency = 1
        lbl.Text = name
        lbl.TextColor3 = Color3.new(1,1,1)
        lbl.TextSize = 12
        lbl.Font = Enum.Font.SourceSansBold
        lbl.TextStrokeTransparency = 0.2
        lbl.Parent = bill
        table.insert(espObjects, bill)
    end
end

-- Main Loop
local hrp, hum
local scan = 0
RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    hrp = char:FindFirstChild("HumanoidRootPart")
    hum = char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    hum.WalkSpeed = settings.speed

    local pos = hrp.Position
    local range = 200 -- bán kính ESP giới hạn level

    -- ESP: chỉ key, két sắt, quái Petapeta trong bán kính
    if settings.esp then
        scan = scan + 1
        if scan % 3 == 0 then
            clearESP()
            -- Quái Petapeta (tên chính xác hoặc chứa "petapeta")
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and (obj.Name:lower():find("petapeta") or obj.Name:lower():find("monster")) then
                    local dist = (obj:FindFirstChild("HumanoidRootPart") and obj.HumanoidRootPart.Position or obj:FindFirstChild("Handle") and obj.Handle.Position)
                    if dist and (dist - pos).Magnitude < range then
                        for _, part in pairs(obj:GetDescendants()) do
                            if part:IsA("BasePart") then createESP(part, Color3.new(1,0.5,0), "Petapeta") end
                        end
                    end
                end
            end
            -- Chìa khóa (key)
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") and part.Name:lower():find("key") then
                    if (part.Position - pos).Magnitude < range then
                        createESP(part, Color3.new(0,0.8,1), "Chìa khóa")
                    end
                end
            end
            -- Két sắt (chest, safe, locker, cabinet, vault, két)
            for _, part in pairs(workspace:GetDescendants()) do
                if part:IsA("BasePart") and (part.Name:lower():find("két") or part.Name:lower():find("chest") or part.Name:lower():find("safe") or part.Name:lower():find("locker") or part.Name:lower():find("cabinet") or part.Name:lower():find("vault")) then
                    if (part.Position - pos).Magnitude < range then
                        createESP(part, Color3.new(0,1,0), "Két sắt")
                    end
                end
            end
        end
    else
        clearESP()
    end

    -- Aura: auto nhặt xu + mở cửa/tủ (bán kính 18)
    if settings.aura and hrp then
        for _, obj in pairs(workspace:GetDescendants()) do
            -- Nhặt xu (coin, money, gold, gem)
            if obj:IsA("BasePart") then
                local n = obj.Name:lower()
                if n:find("coin") or n:find("money") or n:find("gold") or n:find("gem") or n:find("diamond") then
                    if (obj.Position - pos).Magnitude < 18 then
                        firetouchinterest(hrp, obj, 0)
                        firetouchinterest(hrp, obj, 1)
                    end
                end
            end
            -- Mở cửa/tủ bằng ClickDetector
            if obj:IsA("ClickDetector") and obj.Parent then
                local p = obj.Parent
                local target = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if target and (target.Position - pos).Magnitude < 18 then
                    obj.MouseClick:Fire(LocalPlayer)
                end
            end
            -- Mở bằng ProximityPrompt
            if obj:IsA("ProximityPrompt") and obj.Parent then
                local p = obj.Parent
                local posObj = p:IsA("BasePart") and p or (p:FindFirstChild("Handle") and p.Handle)
                if posObj and (posObj.Position - pos).Magnitude < 18 then
                    obj.HoldEnd:Fire(LocalPlayer)
                end
            end
        end
    end
end)

-- F1 toggle menu thu nhỏ/mở rộng
UserInputService.InputBegan:Connect(function(inp, gp)
    if gp then return end
    if inp.KeyCode == Enum.KeyCode.F1 then
        if main.Visible then
            toggleExpand()
        else
            main.Visible = true
        end
    end
end)
